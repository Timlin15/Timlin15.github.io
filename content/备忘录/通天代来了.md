登录自己的账号：
```bash
# codex
mkdir ~/.codex-l
CODEX_HOME=~/.codex-l codex login --device-auth
alias codex-l='CODEX_HOME=~/.codex-l codex'
echo "alias codex-l='VISUAL=vim CODEX_HOME=~/.codex-l codex --sandbox danger-full-access'" >> ~/.bashrc

# claude
mkdir ~/.claude-l
CLAUDE_CONFIG_DIR=~/.claude-l claude /login
alias cc-l='CLAUDE_CONFIG_DIR=~/.claude-l claude'
echo "alias cc-l='CLAUDE_CONFIG_DIR=~/.claude-l claude --dangerously-skip-permissions'" >> ~/.bashrc

# 取消其他中转站 API
unset ANTHROPIC_BASE_URL ANTHROPIC_AUTH_TOKEN
```
